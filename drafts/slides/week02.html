JavaScript สำหรับผู้ที่มาจาก C และ Java

สี่สิบห้านาทีนี้อธิบายสี่เรื่องที่เพิ่งเจอในกล่องทดลอง ด้วยแนวคิดเดียว

เมื่อกี้เจออะไรมา

สี่คำถามที่ยังไม่ได้ตอบ

  • push คืนเลข 4 แทนที่จะคืนอาร์เรย์ และอาร์เรย์เดิมเปลี่ยนไปด้วย
  • ออบเจกต์สองตัวที่ข้างในเหมือนกันทุกอย่าง เทียบแล้วได้ false
  • แก้ จุดC แล้ว จุดA เปลี่ยนตาม ทั้งที่ไม่ได้แตะ
  • const แล้ว push ได้ แต่กำหนดค่าใหม่ไม่ได้

ทั้งสี่ข้อนี้มีคำตอบร่วมกันข้อเดียว

แนวคิดเดียวที่อธิบายได้ทั้งหมด

ใน C ตัวแปรคือกล่องที่มีค่าอยู่ข้างใน
ใน JavaScript บางกล่องเก็บแค่ที่อยู่

ภาษา C

กำหนดค่า struct คือการคัดลอกทั้งก้อน

// C
Point a = {1, 2};
Point b = a;       // คัดลอก
b.x = 99;

printf("%d", a.x); // 1

JavaScript

ออบเจกต์ไม่ได้อยู่ในกล่อง กล่องเก็บแค่ลูกศร

// JavaScript
const a = { x: 1, y: 2 };
const b = a;       // คัดลอกลูกศร
b.x = 99;

console.log(a.x);  // 99

คำถามที่ 2 และ 3

เทียบด้วย === คือเทียบว่าลูกศรชี้ที่เดียวกันไหม

const จุดA = { x: 1, y: 2 };
const จุดB = { x: 1, y: 2 };   // คนละก้อน
const จุดC = จุดA;             // ก้อนเดียวกัน

console.log(จุดA === จุดB);
console.log(จุดA === จุดC);
ข้างในเหมือนกันไม่พอ ต้องเป็นก้อนเดียวกัน

คำถามที่ 4

const ล็อกกล่อง ไม่ได้ล็อกของที่กล่องชี้ไป

const รายการ = ['ก', 'ข'];

รายการ.push('ค');   // แก้ของที่ลูกศรชี้ไป ผ่าน
console.log(รายการ);

try { รายการ = ['ง']; }  // เปลี่ยนลูกศร ไม่ผ่าน
catch (e) { console.log('error:', e.message); }
อยากล็อกของข้างในต้องใช้ Object.freeze ซึ่งคนละเรื่องกัน

คำถามที่ 1

เมธอดของอาร์เรย์มีสองพวก และปนกันไม่ได้

แก้อาร์เรย์เดิม คืนค่าอื่นไม่แตะของเดิม คืนอาร์เรย์ใหม่
push · pop · shift · unshift
splice · sort · reverse		 concat · slice · map · filter
reduce · toSorted · toReversed
push คืนความยาวใหม่ ไม่ใช่อาร์เรย์ จึงต่อ .map() ท้ายไม่ได้ ต่อกันเป็นลูกโซ่ได้ และของเดิมยังอยู่ครบ
ในแล็บนี้ ฟังก์ชันที่รับอาร์เรย์หรือออบเจกต์เข้ามา ห้ามแก้ของที่รับเข้ามา ให้คืนของใหม่เสมอ

เรื่องที่ห้า ที่ยังไม่ได้ถาม

มีตัวเปรียบเทียบสองตัว ตัวหนึ่งแปลงชนิดให้เงียบ ๆ

console.log("5 == '5'    ", 5 == '5');
console.log("5 === '5'   ", 5 === '5');
console.log("0 == ''     ", 0 == '');
console.log("[] == false ", [] == false);
ใน C การเทียบ int กับ char* แบบนี้คอมไพเลอร์ฟ้องตั้งแต่แรก

   นับเป็นตัวคั่นเดียว */
int countWords(const char *s) {
  int count = 0, inWord = 0;
  for (int i = 0; s[i]; i++) {
    if (s[i] == ' ') inWord = 0;
    else if (!inWord) {
      inWord = 1; count++;
    }
  }
  return count;
}
